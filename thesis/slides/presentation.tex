\textbf{Phát hiện bất thường qua MSE:}
$$\text{MSE} = \frac{1}{n}\sum_{i=1}^{n}(x_i - \hat{x}_i)^2$$

\vspace{0.2cm}
\begin{tabular}{ll}
\toprule
\textbf{MSE} & \textbf{Mức độ} \\
\midrule
$\leq$ 1,2093 & Normal \\
1,2093 – 1,5 & WARNING \\
$>$ 1,5 & \textcolor{alertred}{\textbf{CRITICAL}} \\
\bottomrule
\end{tabular}
